CREATE TABLE Author(
  firstname VARCHAR(255),
  surname VARCHAR(255),
  birthdate DATE,
  age_death INT,
  PRIMARY KEY(firstname, surname)
  );

CREATE TABLE Book(
  isbn CHAR(13),
  title VARCHAR(255),
  pages INT,
  a_name VARCHAR(255) NOT NULL,
  a_surname VARCHAR(255) NOT NULL,
  FOREIGN KEY (a_name, a_surname)
  REFERENCES Author(firstname,surname),
  PRIMARY KEY (isbn)
  );

  g_name VARCHAR(255) PRIMARY KEY,

  );

CREATE TABLE Has_Genre(
  b_isbn CHAR(13),
  g_name VARCHAR(255),
  FOREIGN KEY (b_isbn)
  REFERENCES Book(isbn),
  FOREIGN KEY (g_name) REFERENCES Genre(g_name),
  PRIMARY KEY (b_isbn, g_name)
  );

CREATE TABLE Reader(
  rid INT,
  firstname VARCHAR(255) NOT NULL,
  surname VARCHAR(255) NOT NULL,
  favorite_b VARCHAR(255),
  FOREIGN KEY (favorite_b) REFERENCES Book(isbn),
  PRIMARY KEY (rid)
  );

CREATE TABLE Lent_To(
  rid INT NOT NULL ,
  isbn CHAR(13) PRIMARY KEY,
  borrowing_date DATE,
  FOREIGN KEY (isbn) REFERENCES Book(isbn),
  FOREIGN KEY (rid) REFERENCES Reader(rid)
  );

CREATE TABLE Delay(
  original_date DATE ,
  real_date DATE ,
  rid INT ,
  FOREIGN KEY (rid)
  REFERENCES Reader(rid),
  PRIMARY KEY (rid, real_date, original_date)
  );
